  const pageSize = 6;

  let page = 1;

  const search = document.getElementById("search");

  const range = document.getElementById("range");

  const sort = document.getElementById("sort");

  const grid = document.getElementById("sessions");

  const results = document.getElementById("results");

  const empty = document.getElementById("empty");

  const previous = document.getElementById("previous");

  const next = document.getElementById("next");

  const pageInfo = document.getElementById("pageInfo");

  function render() {

    const query = search.value.trim().toLowerCase();

    const numericQuery = query.match(/^(?:s|session\s*)?0*(\d+)$/);

    const bounds = range.value

      ? range.value.split("-").map(Number)

      : null;

    const filtered = sessions.filter(([number, title]) => {

      const matchesSearch = !query || (

        numericQuery

          ? number === Number(numericQuery[1])

          : title.toLowerCase().includes(query)

      );

      const matchesRange = !bounds ||

        (number >= bounds[0] && number <= bounds[1]);

      return matchesSearch && matchesRange;

    });

    filtered.sort((a, b) =>

      sort.value === "newest" ? b[0] - a[0] : a[0] - b[0]

    );

    const pages = Math.max(1, Math.ceil(filtered.length / pageSize));

    page = Math.min(page, pages);

    const start = (page - 1) * pageSize;

    const visible = filtered.slice(start, start + pageSize);

    grid.replaceChildren();

    visible.forEach(([number, title, id]) => {

      const link = document.createElement("a");

      link.className = "session";

      link.href = "https://drive.google.com/open?id=" + id;

      link.target = "_top";

      const badge = document.createElement("span");

      badge.className = "number";

      badge.textContent = "Session " + String(number).padStart(2, "0");

      const heading = document.createElement("h2");

      heading.className = "title";

      heading.textContent = title;

      const read = document.createElement("span");

      read.className = "read";

      read.textContent = "Read transcript";

      const arrow = document.createElement("span");

      arrow.textContent = "→";

      arrow.setAttribute("aria-hidden", "true");

      read.append(arrow);

      link.append(badge, heading, read);

      grid.append(link);

    });

    results.textContent = filtered.length

      ? `Showing ${start + 1}–${start + visible.length} of ${filtered.length} transcripts`

      : "0 matching transcripts";

    empty.hidden = filtered.length !== 0;

    pageInfo.textContent = `Page ${page} of ${pages}`;

    previous.disabled = page === 1;

    next.disabled = page === pages;

  }

  function reset() {

    page = 1;

    render();

  }

  search.addEventListener("input", reset);

  range.addEventListener("change", reset);

  sort.addEventListener("change", reset);

  previous.addEventListener("click", () => {

    if (page > 1) {

      page--;

      render();

    }

  });

  next.addEventListener("click", () => {

    page++;

    render();

  });
